### 1.1 application.yml 注释规范

> [⬆ 返回本章导航](#一配置详解) | [⬆ 返回目录导航](#目录导航)

```yaml
# ==============================================
# 直播电商系统配置文件
# 版本：v1.0
# 更新：2024-01-01
# ==============================================

# 应用名称
spring:
  application:
    name: live-stream

# ==============================================
# 服务配置
# ==============================================
server:
  port: 8080                           # HTTP服务端口
  servlet:
    context-path: /api                  # 上下文路径
  tomcat:
    threads:
      max: 200                          # 最大工作线程数
      min-spare: 10                     # 最小空闲线程数

# ==============================================
# 数据源配置
# ==============================================
  datasource:
    url: jdbc:mysql://localhost:3306/live_stream?useUnicode=true&characterEncoding=utf8
    username: root                      # 数据库用户名
    password: ${DB_PASSWORD}            # 密码（从环境变量读取）
    driver-class-name: com.mysql.cj.jdbc.Driver  # MySQL驱动
    hikari:
      pool-name: LiveStreamHikariPool   # 连接池名称
      maximum-pool-size: 20              # 最大连接数
      minimum-idle: 5                    # 最小空闲连接
      connection-timeout: 30000          # 连接超时（ms）
      idle-timeout: 600000               # 空闲超时（ms）
      max-lifetime: 1800000              # 最大生存时间（ms）

# ==============================================
# Redis配置
# ==============================================
  data:
    redis:
      host: localhost                    # Redis服务器地址
      port: 6379                         # Redis端口
      password: ${REDIS_PASSWORD}        # 密码（无密码留空）
      database: 0                        # 数据库索引
      lettuce:
        pool:
          max-active: 20                 # 最大连接数
          max-idle: 10                   # 最大空闲连接
          min-idle: 5                    # 最小空闲连接

# ==============================================
# 日志配置
# ==============================================
logging:
  level:
    root: INFO                           # 根日志级别
    com.live.stream: DEBUG               # 业务日志级别
  pattern:
    console: '%d{yyyy-MM-dd HH:mm:ss} [%thread] %-5level %logger{36} - %msg%n'
```

### 1.3 logback-spring.xml 注释规范

> [⬆ 返回本章导航](#一配置详解) | [⬆ 返回目录导航](#目录导航)

```xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    <!-- 定义日志文件存放路径 -->
    <property name="LOG_PATH" value="./logs"/>
    
    <!-- 控制台输出格式 -->
    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder>
            <!-- 输出格式：[时间][线程名][级别][类名] 消息内容 -->
            <pattern>%d{yyyy-MM-dd HH:mm:ss}[%thread][%level][%logger{36}] %msg%n</pattern>
            <charset>UTF-8</charset>
        </encoder>
    </appender>

    <!-- 文件输出配置 -->
    <appender name="FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
        <file>${LOG_PATH}/application.log</file>        <!-- 日志文件路径 -->
        <rollingPolicy class="ch.qos.logback.core.rolling.TimeBasedRollingPolicy">
            <!-- 按日期滚动，每天一个新文件 -->
            <fileNamePattern>${LOG_PATH}/application.%d{yyyy-MM-dd}.log</fileNamePattern>
            <maxHistory>30</maxHistory>                  <!-- 保留天数 -->
        </rollingPolicy>
    </appender>

    <!-- 根日志级别 -->
    <root level="INFO">
        <appender-ref ref="CONSOLE"/>
        <appender-ref ref="FILE"/>
    </root>
</configuration>
```

 * 用户Mapper接口

    @Select("SELECT * FROM t_user WHERE username = #{username}")

    /**
     * 插入用户记录
     *
     * @param user 用户实体（id除外，主键自动生成）

    @Insert("INSERT INTO t_user(username, password, phone, create_time) " +

    @Options(useGeneratedKeys = true, keyProperty = "id")
    int insert(User user);
}

### 4.2 XML 映射文件注释规范

> [⬆ 返回本章导航](#四mapper-接口与-xml-注释规范) | [⬆ 返回目录导航](#目录导航)

```xml
<?xml version="1.0" encoding="UTF-8" ?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.live.stream.product.mapper.ProductMapper">

    <!-- 商品结果映射 -->
    <resultMap id="ProductResultMap" type="com.live.stream.product.entity.Product">
        <!-- 主键ID -->
        <id property="id" column="id"/>
        <!-- 商品名称 -->
        <result property="name" column="name"/>
        <!-- 商品价格（单位：分） -->
        <result property="price" column="price"/>
        <!-- 库存数量 -->
        <result property="stock" column="stock"/>
        <!-- 分类ID -->
        <result property="categoryId" column="category_id"/>
        <!-- 创建时间 -->
        <result property="createTime" column="create_time"/>
        <!-- 更新时间 -->
        <result property="updateTime" column="update_time"/>
    </resultMap>

    <!-- 分页查询商品列表 -->
    <select id="selectProductPage" resultMap="ProductResultMap">
        SELECT id, name, price, stock, category_id, create_time, update_time
        FROM t_product
        WHERE del_flag = 0
        <if test="categoryId != null">
            AND category_id = #{categoryId}
        </if>
        <if test="keyword != null and keyword != ''">
            AND name LIKE CONCAT('%', #{keyword}, '%')
        </if>
        ORDER BY create_time DESC
        LIMIT #{offset}, #{pageSize}
    </select>

    <!--：根据ID查询商品详情 -->
    <select id="selectProductDetail" resultType="ProductDetailVO">
        SELECT p.*, c.name AS category_name
        FROM t_product p
        LEFT JOIN t_category c ON p.category_id = c.id
        WHERE p.id = #{id} AND p.del_flag = 0
    </select>
</mapper>
```

# ============================================================================
# 脚本名称: fix-openfeign-rebuild.ps1
# 功能描述: 修复 OpenFeign 接口重构后的编译问题
# 使用场景: 当 Feign 接口路径变更后，自动扫描并修复调用方引用
# 前置条件:
#   1. JDK 21 已安装
#   2. PowerShell 5.1 或更高版本
# 使用方法:
#   1. 直接双击运行
#   2. 或在终端执行: powershell -ExecutionPolicy Bypass -File fix-openfeign-rebuild.ps1
# 作者: 刘武贵
# 创建时间: 2024-01-01
# ============================================================================

# 定义要扫描的模块列表
$modules = @(
    "live-stream-order",
    "live-stream-payment",
    "live-stream-inventory"
)

# 定义旧的 Feign 接口包路径（需替换）
$oldPackage = "com.live.stream.feign.old"
# 定义新的 Feign 接口包路径（替换为）
$newPackage = "com.live.stream.feign.new"

# 遍历每个模块处理
foreach ($module in $modules) {
    Write-Host "[处理] $module ..." -ForegroundColor Cyan
    
    # 1. 扫描 Java 源文件中的旧包引用
    $files = Get-ChildItem -Path "./$module/src" -Filter "*.java" -Recurse
    
    # 2. 替换包引用
    foreach ($file in $files) {
        $content = Get-Content $file.FullName -Raw
        if ($content -match $oldPackage) {
            $newContent = $content -replace $oldPackage, $newPackage
            Set-Content -Path $file.FullName -Value $newContent
            Write-Host "  已修复: $($file.Name)" -ForegroundColor Yellow
        }
    }
}

```sql
-- =============================================
-- 存储过程: sp_batch_update_user_status
-- 功能描述: 批量更新用户状态
-- 参数说明:
--   IN p_user_ids VARCHAR(1000) - 用户ID列表，逗号分隔
--   IN p_status TINYINT - 目标状态
--   OUT p_affected_rows INT - 影响的行数
-- 返回值: 无
-- 使用示例:
--   CALL sp_batch_update_user_status('1,2,3,4,5', 1, @rows);
--   SELECT @rows;
--
-- 作者: 刘武贵
-- 创建时间: 2024-01-01
-- =============================================

DELIMITER //

DROP PROCEDURE IF EXISTS sp_batch_update_user_status//

CREATE PROCEDURE sp_batch_update_user_status(
    IN p_user_ids VARCHAR(1000),     -- 用户ID列表
    IN p_status TINYINT,             -- 目标状态
    OUT p_affected_rows INT          -- 影响行数
)
BEGIN
    -- 声明变量
    DECLARE v_sql VARCHAR(2000);
    DECLARE v_temp_id VARCHAR(50);
    DECLARE v_done INT DEFAULT FALSE;
    
    -- 声明游标
    DECLARE cur CURSOR FOR SELECT id FROM t_user WHERE FIND_IN_SET(id, p_user_ids);
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
    
    -- 初始化计数器
    SET p_affected_rows = 0;
    
    -- 开启事务
    START TRANSACTION;
    
    -- 打开游标
    OPEN cur;
    
    -- 循环处理每个用户
    read_loop: LOOP
        FETCH cur INTO v_temp_id;
        
        IF v_done THEN
            LEAVE read_loop;
        END IF;
        
        -- 更新用户状态
        UPDATE t_user 
        SET status = p_status, update_time = NOW()
        WHERE id = v_temp_id AND del_flag = 0;
        
        -- 计数器加一
        SET p_affected_rows = p_affected_rows + ROW_COUNT();
    END LOOP;
    
    -- 关闭游标
    CLOSE cur;
    
    -- 提交事务
    COMMIT;
END //

DELIMITER ;
```

```lua
-- =============================================
-- 脚本名称: rate_limit.lua
-- 功能描述: 滑动窗口限流算法实现
-- 使用场景: API 接口限流、用户行为限制
-- 参数说明:
--   KEYS[1] - 限流 key（如: rate_limit:user:123）
--   ARGV[1] - 窗口大小（毫秒）
--   ARGV[2] - 最大请求数
-- 返回值:
--   1 - 允许通过
--   0 - 被限流
-- 使用示例:
--   EVAL rate_limit.lua 1 "rate_limit:user:123" 60000 100
--
-- 作者: 刘武贵
-- 创建时间: 2024-01-01
-- =============================================

-- 获取限流 key
local key = KEYS[1]

-- 获取配置参数
local window = tonumber(ARGV[1])     -- 窗口大小（毫秒）
local max_requests = tonumber(ARGV[2])  -- 最大请求数

-- 获取当前时间戳（毫秒）
local now = tonumber(ARGV[3]) or redis.call('TIME')[1] * 1000

-- 计算窗口起始时间
local window_start = now - window

-- 1. 移除窗口外的历史记录
redis.call('ZREMRANGEBYSCORE', key, '-inf', window_start)

-- 2. 统计当前窗口内的请求数
local current_count = redis.call('ZCARD', key)

-- 3. 判断是否超过限制
if current_count < max_requests then
    -- 未超过限制：记录本次请求
    redis.call('ZADD', key, now, now .. '-' .. math.random())
    
    -- 设置过期时间（防止 key 堆积）
    redis.call('PEXPIRE', key, window)
    
    return 1  -- 允许通过
else
    return 0  -- 被限流
end
```

> [⬆ 返回本章导航](#十一lua-脚本注释规范) | [⬆ 返回目录导航](#目录导航)

```lua
-- =============================================
-- 脚本名称: auth.lua
-- 功能描述: JWT 令牌验证中间件
-- 适用场景: API 网关请求鉴权
-- 使用方式: 在 Nginx 配置中引用
--   access_by_lua_file /path/to/auth.lua;
--
-- 功能说明:
--   1. 从请求头获取 JWT 令牌
--   2. 验证令牌签名和有效期
--   3. 将用户信息注入请求头传递给后端
--   4. 验证失败返回 401 Unauthorized
--
-- 作者: 刘武贵
-- 创建时间: 2024-01-01
-- =============================================

local cjson = require('cjson')
local jwt = require('resty.jwt')

-- 配置常量
local JWT_SECRET = os.getenv('JWT_SECRET') or 'your-secret-key'  -- JWT 密钥
local TOKEN_HEADER = 'Authorization'                            -- 令牌头名称
local TOKEN_PREFIX = 'Bearer '                                   -- 令牌前缀

-- ------------------------------------------
-- 函数: 从请求头获取令牌
-- ------------------------------------------
local function get_token_from_header()
    local auth_header = ngx.req.get_headers()[TOKEN_HEADER]
    if not auth_header then
        return nil, 'Missing Authorization header'
    end
    
    -- 检查令牌格式
    if string.sub(auth_header, 1, #TOKEN_PREFIX) ~= TOKEN_PREFIX then
        return nil, 'Invalid Authorization format'
    end
    
    -- 提取令牌
    return string.sub(auth_header, #TOKEN_PREFIX + 1)
end

-- ------------------------------------------
-- 函数: 验证令牌
-- ------------------------------------------
local function validate_token(token)
    local success, result = pcall(function()
        return jwt:verify(JWT_SECRET, token)
    end)
    
    if not success then
        return nil, 'Token verification failed: ' .. tostring(result)
    end
    
    if not result.valid then
        return nil, 'Invalid token: ' .. (result.reason or 'unknown')
    end
    
    return result payload
end

-- ------------------------------------------
-- 主逻辑
-- ------------------------------------------
local token, err = get_token_from_header()

if not token then
    ngx.status = ngx.HTTP_UNAUTHORIZED
    ngx.say(cjson.encode({
        code = 401,
        message = err or 'Unauthorized'
    }))
    return ngx.exit(ngx.HTTP_UNAUTHORIZED)
end

local payload, err = validate_token(token)

if not payload then
    ngx.status = ngx.HTTP_UNAUTHORIZED
    ngx.say(cjson.encode({
        code = 401,
        message = err or 'Invalid token'
    }))
    return ngx.exit(ngx.HTTP_UNAUTHORIZED)
end

-- 将用户信息注入请求头
ngx.req.set_header('X-User-Id', payload.sub)
ngx.req.set_header('X-User-Name', payload.name)
ngx.req.set_header('X-User-Role', payload.role)

-- 放行请求
```

### 11.3 OpenResty 限流配置注释规范

> [⬆ 返回本章导航](#十一lua-脚本注释规范) | [⬆ 返回目录导航](#目录导航)

```nginx
# =============================================
# rate_limit.conf - OpenResty 限流配置
# =============================================
# 功能描述: 基于 Redis + Lua 的滑动窗口限流
# 适用场景: API 接口限流、CC 攻击防护
#
# 配置说明:
#   - 按用户 ID 限流（key: rate_limit:user:{user_id}）
#   - 窗口大小: 60 秒
#   - 最大请求数: 100 次/分钟
#   - 超出限制返回 429 Too Many Requests
#
# 使用方法:
#   在 server 或 location 中引入: include rate_limit.conf;
#
# 作者: 刘武贵
# 创建时间: 2024-01-01
# =============================================

# 限流区域定义（共享内存）
# 语法: lua_shared_dict <name> <size>
# 用途: 存储限流计数器（使用 Redis 替代，此处仅作示例）
lua_shared_dict rate_limit_store 10m;

# 限流配置常量
# 通过 set_by_lua 初始化配置
set_by_lua $rate_limit_config '
    return cjson.encode({
        window = 60000,      -- 窗口大小（毫秒）
        max_requests = 100,  -- 最大请求数
        key_prefix = "rate_limit:user:"  -- Redis key 前缀
    })
';

# 限流逻辑（access_by_lua 阶段执行）
access_by_lua_block {
    local config = cjson.decode(ngx.var.rate_limit_config)
    
    -- 获取用户标识（优先用户ID，其次IP）
    local user_id = ngx.req.get_headers()['X-User-Id'] or ngx.var.remote_addr
    local key = config.key_prefix .. user_id
    
    -- 调用限流脚本
    local rate_limit = require('rate_limit')
    local allowed, remaining = rate_limit.check(key, config.window, config.max_requests)
    
    -- 设置响应头
    ngx.header['X-RateLimit-Limit'] = config.max_requests
    ngx.header['X-RateLimit-Remaining'] = remaining
    
    if not allowed then
        ngx.header['Retry-After'] = math.ceil(config.window / 1000)
        ngx.status = 429
        ngx.say('{"code": 429, "message": "请求过于频繁，请稍后再试"}')
        return ngx.exit(429)
    end
}
```
